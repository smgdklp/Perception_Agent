// ============================================================
// 方式1：直接保存 BGRA（Alpha 通道可能被当作数据）
// ============================================================
bool SaveBmp_BGRA(const std::string& filename, const cv::Mat& mat) {
    if (mat.empty() || mat.type() != CV_8UC4) return false;

    int width = mat.cols;
    int height = mat.rows;
    int stride = width * 4;

    BITMAPFILEHEADER fh = {0};
    fh.bfType = 0x4D42;
    fh.bfSize = sizeof(BITMAPFILEHEADER) + sizeof(BITMAPINFOHEADER) + stride * height;
    fh.bfOffBits = sizeof(BITMAPFILEHEADER) + sizeof(BITMAPINFOHEADER);

    BITMAPINFOHEADER ih = {0};
    ih.biSize = sizeof(BITMAPINFOHEADER);
    ih.biWidth = width;
    ih.biHeight = -height;
    ih.biPlanes = 1;
    ih.biBitCount = 32;
    ih.biCompression = BI_RGB;
    ih.biSizeImage = 0;

    std::ofstream file(filename, std::ios::binary);
    if (!file.is_open()) return false;

    file.write(reinterpret_cast<const char*>(&fh), sizeof(fh));
    file.write(reinterpret_cast<const char*>(&ih), sizeof(ih));
    file.write(reinterpret_cast<const char*>(mat.data), stride * height);
    file.close();
    return true;
}

// ============================================================
// 方式2：BGRA → BGR（去掉 Alpha，转换为 24 位 BMP）
// ============================================================
bool SaveBmp_BGR(const std::string& filename, const cv::Mat& mat) {
    if (mat.empty() || mat.type() != CV_8UC4) return false;

    int width = mat.cols;
    int height = mat.rows;
    int stride = width * 3;  // BGR 每行 3 字节

    // 转换 BGRA → BGR
    std::vector<unsigned char> bgrData(stride * height);
    for (int y = 0; y < height; ++y) {
        const unsigned char* src = mat.ptr<unsigned char>(y);
        unsigned char* dst = bgrData.data() + y * stride;
        for (int x = 0; x < width; ++x) {
            dst[x * 3 + 0] = src[x * 4 + 0];  // B
            dst[x * 3 + 1] = src[x * 4 + 1];  // G
            dst[x * 3 + 2] = src[x * 4 + 2];  // R
            // 忽略 Alpha
        }
    }

    BITMAPFILEHEADER fh = {0};
    fh.bfType = 0x4D42;
    fh.bfSize = sizeof(BITMAPFILEHEADER) + sizeof(BITMAPINFOHEADER) + stride * height;
    fh.bfOffBits = sizeof(BITMAPFILEHEADER) + sizeof(BITMAPINFOHEADER);

    BITMAPINFOHEADER ih = {0};
    ih.biSize = sizeof(BITMAPINFOHEADER);
    ih.biWidth = width;
    ih.biHeight = -height;
    ih.biPlanes = 1;
    ih.biBitCount = 24;
    ih.biCompression = BI_RGB;
    ih.biSizeImage = 0;

    std::ofstream file(filename, std::ios::binary);
    if (!file.is_open()) return false;

    file.write(reinterpret_cast<const char*>(&fh), sizeof(fh));
    file.write(reinterpret_cast<const char*>(&ih), sizeof(ih));
    file.write(reinterpret_cast<const char*>(bgrData.data()), stride * height);
    file.close();
    return true;
}

// ============================================================
// 方式3：BGRA → BGR，但 Alpha 强制设为 255（32 位）
// ============================================================
bool SaveBmp_BGRA_FixAlpha(const std::string& filename, const cv::Mat& mat) {
    if (mat.empty() || mat.type() != CV_8UC4) return false;

    int width = mat.cols;
    int height = mat.rows;
    int stride = width * 4;

    // 复制数据，强制 Alpha = 255
    std::vector<unsigned char> data(stride * height);
    for (int y = 0; y < height; ++y) {
        const unsigned char* src = mat.ptr<unsigned char>(y);
        unsigned char* dst = data.data() + y * stride;
        for (int x = 0; x < width; ++x) {
            dst[x * 4 + 0] = src[x * 4 + 0];  // B
            dst[x * 4 + 1] = src[x * 4 + 1];  // G
            dst[x * 4 + 2] = src[x * 4 + 2];  // R
            dst[x * 4 + 3] = 255;             // Alpha 强制 255
        }
    }

    BITMAPFILEHEADER fh = {0};
    fh.bfType = 0x4D42;
    fh.bfSize = sizeof(BITMAPFILEHEADER) + sizeof(BITMAPINFOHEADER) + stride * height;
    fh.bfOffBits = sizeof(BITMAPFILEHEADER) + sizeof(BITMAPINFOHEADER);

    BITMAPINFOHEADER ih = {0};
    ih.biSize = sizeof(BITMAPINFOHEADER);
    ih.biWidth = width;
    ih.biHeight = -height;
    ih.biPlanes = 1;
    ih.biBitCount = 32;
    ih.biCompression = BI_RGB;
    ih.biSizeImage = 0;

    std::ofstream file(filename, std::ios::binary);
    if (!file.is_open()) return false;

    file.write(reinterpret_cast<const char*>(&fh), sizeof(fh));
    file.write(reinterpret_cast<const char*>(&ih), sizeof(ih));
    file.write(reinterpret_cast<const char*>(data.data()), stride * height);
    file.close();
    return true;
}

// ============================================================
// 方式4：BGRA → RGB（交换 R 和 B）
// ============================================================
bool SaveBmp_RGB(const std::string& filename, const cv::Mat& mat) {
    if (mat.empty() || mat.type() != CV_8UC4) return false;

    int width = mat.cols;
    int height = mat.rows;
    int stride = width * 3;

    std::vector<unsigned char> rgbData(stride * height);
    for (int y = 0; y < height; ++y) {
        const unsigned char* src = mat.ptr<unsigned char>(y);
        unsigned char* dst = rgbData.data() + y * stride;
        for (int x = 0; x < width; ++x) {
            dst[x * 3 + 0] = src[x * 4 + 2];  // R
            dst[x * 3 + 1] = src[x * 4 + 1];  // G
            dst[x * 3 + 2] = src[x * 4 + 0];  // B
        }
    }

    BITMAPFILEHEADER fh = {0};
    fh.bfType = 0x4D42;
    fh.bfSize = sizeof(BITMAPFILEHEADER) + sizeof(BITMAPINFOHEADER) + stride * height;
    fh.bfOffBits = sizeof(BITMAPFILEHEADER) + sizeof(BITMAPINFOHEADER);

    BITMAPINFOHEADER ih = {0};
    ih.biSize = sizeof(BITMAPINFOHEADER);
    ih.biWidth = width;
    ih.biHeight = -height;
    ih.biPlanes = 1;
    ih.biBitCount = 24;
    ih.biCompression = BI_RGB;
    ih.biSizeImage = 0;

    std::ofstream file(filename, std::ios::binary);
    if (!file.is_open()) return false;

    file.write(reinterpret_cast<const char*>(&fh), sizeof(fh));
    file.write(reinterpret_cast<const char*>(&ih), sizeof(ih));

    return true;
}

void PrintMatInfo(const cv::Mat& mat) {
    if (mat.empty()) {
        std::cout << "[MatInfo] empty" << std::endl;
        return;
    }
    std::cout << "[MatInfo] " << mat.cols << "x" << mat.rows << ", type=" << mat.type();
    if (mat.type() == CV_8UC4) {
        cv::Vec4b p = mat.at<cv::Vec4b>(0, 0);
        std::cout << ", pixel(0,0)=(" << (int)p[0] << "," << (int)p[1] << "," << (int)p[2] << "," << (int)p[3] << ")";
    }
    std::cout << std::endl;
}
